#!/usr/bin/env python3
"""rec.py KIND SECONDS [fps] [identity|x,y,w,h]: record with put.gd through the CLI (no picker allowed), save a copy,
score motion (putgd-record --motion) and count distinct frames (ffmpeg mpdecimate)."""
import json, os, subprocess, sys, time, re
acc = os.path.expanduser('~/d/probe/acc.py')
def cli(m, p=None, t=60):
    r = subprocess.run([acc, 'cli', m] + ([json.dumps(p)] if p is not None else []) + [str(t)], capture_output=True, text=True)
    return json.loads(r.stdout)
kind, secs = sys.argv[1], float(sys.argv[2]); fps = int(sys.argv[3]) if len(sys.argv) > 3 else 30
target = {'kind': kind}
if len(sys.argv) > 4:
    if kind == 'region': target['rect'] = [int(v) for v in sys.argv[4].split(',')]
    else: target['identity'] = sys.argv[4]
r = cli('recording.prepare', {'target': target, 'format': 'mp4', 'fps': fps, 'adaptiveFps': False, **({'audio': {'system': True}} if os.environ.get('AUDIO') else {})})
if not r.get('ok'): print(json.dumps({'prepare': r.get('error')})); sys.exit(1)
rid = r['result']['recordingId']; t0 = time.time()
while True:
    s = cli('recording.inspect', {'id': rid})['result']
    if s['state'] != 'preparing' or time.time() - t0 > 30: break
    time.sleep(0.3)
pickers = subprocess.run(['pgrep', '-a', '-f', 'share-picker'], capture_output=True, text=True).stdout.strip()
print(json.dumps({'id': rid, 'state': s['state'], 'target': s.get('target'), 'window': s.get('window'), 'canvas': s.get('canvas'), 'error': s.get('error'), 'pickerProcesses': pickers}))
if s['state'] != 'ready': sys.exit(1)
cli('recording.start', {'id': rid}); time.sleep(secs)
cli('recording.stop', {'id': rid})
for _ in range(100):
    s = cli('recording.inspect', {'id': rid})['result']
    if s['state'] == 'reviewing': break
    time.sleep(0.3)
out = os.path.expanduser(f'~/d/rec-{kind}-{int(time.time())}.mp4')
cli('recording.saveCopy', {'id': rid, 'path': out})
for _ in range(200):
    s = cli('recording.inspect', {'id': rid})['result']
    if s['state'] == 'reviewing' and s.get('lastExport', {}).get('status') in ('saved', 'failed'): break
    time.sleep(0.3)
print(json.dumps({'export': s.get('lastExport'), 'durationUs': s.get('durationUs'), 'progress': {k: s.get('progress', {}).get(k) for k in ('framesEncoded', 'framesDropped', 'fps')}}))
cli('recording.cancel', {'id': rid})
motion = subprocess.run(['putgd-record', '--motion', out], capture_output=True, text=True)
print('motion:', (motion.stdout + motion.stderr).strip()[-300:])
probe = subprocess.run(['ffprobe', '-v', 'error', '-select_streams', 'v', '-count_frames', '-show_entries', 'stream=width,height,nb_read_frames,r_frame_rate,duration', '-of', 'json', out], capture_output=True, text=True)
print('ffprobe:', probe.stdout.replace('\n', ' '))
dec = subprocess.run(['ffmpeg', '-nostdin', '-i', out, '-vf', 'mpdecimate=hi=64*4:lo=64:frac=0.1', '-f', 'null', '-'], capture_output=True, text=True)
frames = re.findall(r'frame=\s*(\d+)', dec.stderr)
print('distinct frames (mpdecimate):', frames[-1] if frames else '?', 'file:', out)
