cmake_minimum_required(VERSION 3.24)
project(tree_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4.2 REQUIRED COMPONENTS Core Gui Widgets DBus Test)
add_executable(test_gui_tray_season tests/gui/test_tray_season.cpp src/ui/Ui.cpp src/ui/IconArt.cpp src/ui/Theme.cpp src/ui/WindowButtons.cpp src/ui/TrayCalendar.cpp)
target_include_directories(test_gui_tray_season PRIVATE src tests)
target_link_libraries(test_gui_tray_season PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test)
enable_testing()
add_test(NAME gui_tray_season COMMAND test_gui_tray_season)
set_tests_properties(gui_tray_season PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
